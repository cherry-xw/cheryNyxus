cmake_minimum_required(VERSION 3.16)
project(adv_foundation_host LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(MOONCAKE_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
if(NOT MOONCAKE_SOURCE)
  message(FATAL_ERROR "Set MOONCAKE_SOURCE to the pinned reference checkout")
endif()
add_subdirectory(${MOONCAKE_SOURCE} mooncake)
add_executable(foundation_test foundation_test.cpp ../../../../platform/adv/components/adv_foundation/foundation.cpp ../../../../platform/adv/components/adv_foundation/sd_store.cpp)
target_include_directories(foundation_test PRIVATE stubs ../../../../platform/adv/components/adv_foundation/include)
target_link_libraries(foundation_test PRIVATE mooncake)
enable_testing()
add_test(NAME foundation COMMAND foundation_test)
add_executable(wifi_credentials_test ../wifi_credentials_test.cpp
  ../../../../platform/adv/main/wifi_credentials.cpp
  ../../../../platform/adv/main/wifi_boot.cpp)
target_include_directories(wifi_credentials_test PRIVATE stubs)
target_compile_options(wifi_credentials_test PRIVATE -Wall -Wextra -Werror)
add_test(NAME wifi_credentials COMMAND wifi_credentials_test)
