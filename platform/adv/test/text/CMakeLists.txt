cmake_minimum_required(VERSION 3.16)
project(cheryNyxus_text_host LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 17)
set(LV_CONF_SKIP ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_EXAMPLES ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_DEMOS ON CACHE BOOL "" FORCE)
set(LV_CONF_BUILD_DISABLE_THORVG_INTERNAL ON CACHE BOOL "" FORCE)
if(NOT LVGL_SOURCE)
  message(FATAL_ERROR "Set LVGL_SOURCE to the pinned LVGL 9.2.2 source")
endif()
add_subdirectory(${LVGL_SOURCE} lvgl)
add_executable(text_input_test ../text_input_test.cpp)
target_compile_options(text_input_test PRIVATE -Wall -Wextra -Werror)
add_executable(textarea_test textarea_test.cpp ../../../../platform/adv/main/fonts/adv_font_basic.c)
target_link_libraries(textarea_test PRIVATE lvgl m)
target_compile_options(textarea_test PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_test(NAME text_input COMMAND text_input_test)
add_test(NAME textarea COMMAND textarea_test ${CMAKE_CURRENT_SOURCE_DIR}/../../../../platform/adv/main/fonts/basic-characters.txt)
